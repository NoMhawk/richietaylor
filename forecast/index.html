globalWeatherData = responses[0]; multiModelData = { ecmwf_ifs025: responses[1], gfs_seamless: responses[2], icon_seamless: responses[3], meteofrance_seamless: responses[4] }; document.getElementById('last-updated').innerHTML = ` Updated: ${new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})} (${CONFIG.revision})`;
